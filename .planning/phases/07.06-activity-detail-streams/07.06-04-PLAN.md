       ```python
       @router.get(
           "/{activity_id}/streams",
           response_model=ActivityStreamsResponse,
           summary="Get time-series streams for an activity",
       )
       async def get_activity_streams(
           activity_id: int,
           current_user: Annotated[User, Depends(get_current_active_user)],
           db: Annotated[AsyncSession, Depends(get_db_for_user)],
       ) -> ActivityStreamsResponse:
           activity = await activity_service.get_activity(db, current_user.id, activity_id)
           if activity is None:
               raise HTTPException(status_code=404, detail="Activity not found")

           profile = await get_or_create_profile(db, current_user.id)

           ef = efficiency_factor(activity.normalized_power, activity.average_hr)
           if_val = None
           ftp_ref = None
           if activity.normalized_power and profile and profile.ftp_watts:
               if_val = round(activity.normalized_power / profile.ftp_watts, 3)
               ftp_ref = float(profile.ftp_watts)
           work_kj = None
           if activity.average_power and activity.duration_seconds:
               work_kj = round((activity.average_power * activity.duration_seconds) / 1000, 1)

           return ActivityStreamsResponse(
               activity_id=activity_id,
               has_streams=activity.sample_streams is not None,
               streams=activity.sample_streams,
               headline=StreamsHeadline(
                   tss=activity.tss,
                   normalized_power=activity.normalized_power,
                   intensity_factor=if_val,
                   efficiency_factor=ef,
                   work_kj=work_kj,
                   ftp_reference=ftp_ref,
               ),
           )
       ```
