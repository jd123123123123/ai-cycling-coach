       ```typescript
         )

       }
       ```

    2. Create `frontend/src/hooks/useActivityStreams.ts`:

       ```typescript
       import { useQuery } from '@tanstack/react-query'
       import { fetchActivityStreams } from '../api/activities'
       import type { ActivityStreamsResponse } from '../types/dashboard'

       export function useActivityStreams(activityId: number | null | undefined) {
         return useQuery<ActivityStreamsResponse>({
           queryKey: ['activity-streams', activityId],
           queryFn: () => fetchActivityStreams(activityId as number),
           enabled: activityId != null,
           staleTime: 10 * 60 * 1000, // streams don't change after activity completes
         })
       }
       ```

       The QueryClient already configures retry:false on 401/403/404 (per state.md decision [04-01]) — no override needed here.
