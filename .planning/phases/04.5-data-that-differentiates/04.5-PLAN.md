     ```ts
     import type { ActivityRead, PMCEntry } from '../api/dashboard'

     // Efficiency Factor = NP / avg_HR. Higher = better aerobic economy.
     // Requires both normalized_power and average_hr — returns null if either missing.
     export interface EfficiencyFactorPoint {
       date: string       // ISO started_at
       ef: number         // NP / avg_HR
       sport: ActivityRead['sport']
     }

     export function efficiencyFactorSeries(activities: ActivityRead[]): EfficiencyFactorPoint[] {
       // We only have NP for cycling. Fallback: skip non-cycling in the trend.
       return activities
         .filter((a) => a.sport === 'cycling')
         .filter((a): a is ActivityRead & { normalized_power: number; average_hr: number } => {
           // normalized_power isn't in ActivityRead type yet — add it to dashboard.ts types
           // For now we'll filter defensively.
           // eslint-disable-next-line @typescript-eslint/no-explicit-any
           const raw = a as any
           return typeof raw.normalized_power === 'number' && typeof raw.average_hr === 'number' && raw.average_hr > 0
         })
         .map((a) => ({
           date: a.started_at,
           // eslint-disable-next-line @typescript-eslint/no-explicit-any
           ef: (a as any).normalized_power / (a as any).average_hr,
           sport: a.sport,
         }))
         .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
     }

     // 4-week (28-day) moving average of EF — used as the trend line.
     export function movingAverage<T extends { date: string; ef: number }>(
       points: T[],
       windowDays: number,
     ): Array<{ date: string; value: number | null }> {
       return points.map((p, idx) => {
         const cutoff = new Date(p.date).getTime() - windowDays * 86_400_000
         const within = points.slice(0, idx + 1).filter((q) => new Date(q.date).getTime() >= cutoff)
         if (within.length < 3) return { date: p.date, value: null }
         const sum = within.reduce((s, q) => s + q.ef, 0)
         return { date: p.date, value: sum / within.length }
       })
     }

     // Polarized 3-zone classification per activity using average_hr vs LTHR.
     // Seiler's zones:
     //   Zone 1 (easy): below ventilatory threshold 1 ≈ < 82% LTHR
     //   Zone 2 (moderate): between VT1 and LT ≈ 82% – 97% LTHR (aka "grey zone")
     //   Zone 3 (hard): above LT ≈ > 97% LTHR
     export type PolarizedZone = 'easy' | 'moderate' | 'hard'

     export function classifyPolarizedZone(
       avgHr: number | null,
       lthr: number | null,
     ): PolarizedZone | null {
       if (avgHr == null || lthr == null || lthr <= 0) return null
       const pct = avgHr / lthr
       if (pct < 0.82) return 'easy'
       if (pct <= 0.97) return 'moderate'
       return 'hard'
     }

     export interface TIDWeek {
       weekStart: string                  // ISO yyyy-mm-dd (Monday)
       totals: Record<PolarizedZone, number>  // hours per zone
       totalHours: number
     }

     // Aggregate activities into per-week polarized zone hours, last 8 weeks.
     export function weeklyTID(
       activities: ActivityRead[],
       lthr: number | null,
       weeksBack = 8,
     ): TIDWeek[] {
       const now = new Date()
       // Start of current ISO week (Monday 00:00 local)
       const startOfWeek = new Date(now)
       const day = (startOfWeek.getDay() + 6) % 7  // 0 = Monday
       startOfWeek.setHours(0, 0, 0, 0)
       startOfWeek.setDate(startOfWeek.getDate() - day)

       const weeks: TIDWeek[] = []
       for (let i = weeksBack - 1; i >= 0; i--) {
         const ws = new Date(startOfWeek)
         ws.setDate(startOfWeek.getDate() - i * 7)
         const we = new Date(ws)
         we.setDate(ws.getDate() + 7)
         const bucket: TIDWeek = {
           weekStart: ws.toISOString().slice(0, 10),
           totals: { easy: 0, moderate: 0, hard: 0 },
           totalHours: 0,
         }
         for (const a of activities) {
           const t = new Date(a.started_at).getTime()
           if (t < ws.getTime() || t >= we.getTime()) continue
           const zone = classifyPolarizedZone(a.average_hr, lthr)
           if (zone == null) continue
           const hours = (a.duration_seconds ?? 0) / 3600
           bucket.totals[zone] += hours
           bucket.totalHours += hours
         }
         weeks.push(bucket)
       }
       return weeks
     }

     // Aggregate current week totals for a single donut display.
     export function currentWeekTID(
       activities: ActivityRead[],
       lthr: number | null,
     ): TIDWeek {
       const weeks = weeklyTID(activities, lthr, 1)
       return weeks[weeks.length - 1]!
     }

     // Calendar heatmap data: last 12 weeks × 7 days = 84 cells, each with {date, tss}.
     // PMCEntry has daily_tss per date. Returns cells grouped by week for ECharts.
     export interface HeatmapCell {
       date: string      // yyyy-mm-dd
       tss: number       // daily TSS (0 if rest day)
     }

     export function calendarCells(pmcEntries: PMCEntry[], daysBack = 84): HeatmapCell[] {
       const today = new Date()
       today.setHours(0, 0, 0, 0)
       const map = new Map<string, number>()
       for (const e of pmcEntries) {
         map.set(e.date, e.daily_tss ?? 0)
       }
       const cells: HeatmapCell[] = []
       for (let i = daysBack - 1; i >= 0; i--) {
         const d = new Date(today)
         d.setDate(today.getDate() - i)
         const iso = d.toISOString().slice(0, 10)
         cells.push({ date: iso, tss: map.get(iso) ?? 0 })
       }
       return cells
     }

     // HRV baseline band: rolling mean ± (coefficient of variation × mean).
     // Returns per-day {value, lower, upper} from a 7-day lookback.
     export interface HrvBandPoint {
       value: number | null
       lower: number | null
       upper: number | null
     }

     export function hrvBaselineBand(
       series: Array<number | null>,
       window = 7,
     ): HrvBandPoint[] {
       return series.map((_, i) => {
         const slice = series.slice(Math.max(0, i - window + 1), i + 1)
                            .filter((v): v is number => typeof v === 'number')
         if (slice.length < 3) {
           return { value: series[i] ?? null, lower: null, upper: null }
         }
         const mean = slice.reduce((s, v) => s + v, 0) / slice.length
         const variance = slice.reduce((s, v) => s + (v - mean) ** 2, 0) / slice.length
         const std = Math.sqrt(variance)
         return {
           value: series[i],
           lower: mean - std,
           upper: mean + std,
         }
       })
     }
     ```

    Key uPlot patterns for React 19:
    ```tsx
    import uPlot from 'uplot'
    import 'uplot/dist/uPlot.min.css'
    import { useEffect, useRef } from 'react'

    useEffect(() => {
      if (!ref.current || series.length === 0) return
      const opts: uPlot.Options = {
        width: ref.current.clientWidth,
        height: 280,
        series: [
          {},                                                       // x-axis (timestamps)
          { label: 'Fitness', stroke: '#1c4433', width: 2.25 },    // CTL
          { label: 'Fatigue', stroke: '#b04a0c', width: 1.5, dash: [5, 3] },  // ATL
          { label: 'Form',    stroke: '#143555', width: 2 },        // TSB
        ],
        axes: [
          { stroke: '#757465', grid: { show: false }, font: '11px Geist, sans-serif' },
          { stroke: '#757465', grid: { stroke: '#e2d8c0', width: 0.5, dash: [2, 3] } },
        ],
        hooks: {
          drawClear: [
            (u) => {
              // Paint TSB zone bands on the canvas before any series lines.
              // Assume TSB series is `series[3]`. We use the y-scale of that series.
              const ctx = u.ctx
              const scale = u.scales.y
              if (scale == null) return
              // Zone: at-risk <-10, loading -10..-5, optimal -5..5, fresh 5..15, over >15
              const zones: Array<[number, number, string]> = [
                [-200, -10, 'rgba(138, 29, 29, 0.05)'],   // at-risk
                [-10, 5, 'rgba(20, 53, 85, 0.04)'],        // optimal
                [5, 200, 'rgba(28, 68, 51, 0.04)'],         // fresh
              ]
              for (const [y0, y1, fill] of zones) {
                const yTop = u.valToPos(y1, 'y', true)
                const yBot = u.valToPos(y0, 'y', true)
                ctx.fillStyle = fill
                ctx.fillRect(u.bbox.left, yTop, u.bbox.width, yBot - yTop)
              }
            },
          ],
        },
        cursor: { drag: { x: false }, focus: { prox: 16 } },
        legend: { show: false },
      }
      const data: uPlot.AlignedData = [
        series.map((s) => new Date(s.date).getTime() / 1000),
        series.map((s) => s.ctl),
        series.map((s) => s.atl),
        series.map((s) => s.tsb),
      ]
      const plot = new uPlot(opts, data, ref.current)
      // Resize observer to handle container changes
      const ro = new ResizeObserver(() => {
        if (ref.current) plot.setSize({ width: ref.current.clientWidth, height: 280 })
      })
      ro.observe(ref.current)
      return () => {
        ro.disconnect()
        plot.destroy()
      }
    }, [series])
    ```
