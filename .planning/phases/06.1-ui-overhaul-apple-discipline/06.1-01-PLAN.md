// Framer Motion mapping: stiffness k = (2π/response)², damping c = 2 × damping × √k
// Values precomputed so components stay declarative.
export const springs = {
  smooth:      { type: 'spring' as const, stiffness: 130,  damping: 19 },
  snappy:      { type: 'spring' as const, stiffness: 322,  damping: 32 },

} as const

**Build `frontend/src/components/ui/SideSheet.tsx`** — right drawer primitive (~120-160 LOC). Contract:

```ts
interface SideSheetProps {
  open: boolean
  onClose: () => void
  title?: string
  /** Desktop width. Default 600. Range 540-640 per REDESIGN-BRIEF §3. */
  width?: number
  /** Optional eyebrow label above title, e.g. "Why this?" */
  eyebrow?: string
  children: React.ReactNode
}
```

Behavior requirements:
- Uses `motion/react` (already installed — the 05-02 invariant) AnimatePresence + motion.aside.
- Desktop (>= 768px): slides in from the right, width = props.width ?? 600, full viewport height, positioned `fixed right-0 top-0 bottom-0`. Uses `springs.smooth` from motion.ts.
- Mobile (< 768px): full-screen, slides from right, width 100vw. Same spring.
- Backdrop: `fixed inset-0` with `bg-black/30 backdrop-blur-[6px]`, fades in/out (opacity 0 → 1, duration-short). Clicking backdrop calls `onClose`.
- Escape key: `useEffect` binds `keydown` → if `e.key === 'Escape'` call `onClose`. Cleanup on unmount.
- Focus trap: on open, capture `document.activeElement` as `previouslyFocused`; move focus to the sheet root (set `tabIndex={-1}` on the `<aside>` + call `.focus()` in a mount effect). On close, restore focus to `previouslyFocused`. **Do NOT import `focus-trap-react`** — this is a minimal trap to stay bundle-neutral.
- ARIA: `role="dialog"` + `aria-modal="true"` + `aria-labelledby` pointing at the title element.
- Styling: uses existing editorial palette — background `var(--color-paper-2)`, border-left `1px solid var(--color-rule)`, header area with eyebrow + title + close button (lucide `X` icon, 20px). Uses `--radius-xl` on the left edge only via `border-top-left-radius` + `border-bottom-left-radius` on desktop, no radius on mobile. `--shadow-4` for depth.
- Portal: renders via `createPortal(..., document.body)` so z-index doesn't fight the masthead.
- Language awareness: no copy inside the primitive — all text comes from props (`title`, `eyebrow`) or `children`.

**Build `frontend/src/components/ui/SegmentedTabs.tsx`** (~60-90 LOC). Contract:

```ts
interface SegmentedTabsProps<T extends string> {
  tabs: readonly { id: T; label: string }[]
  value: T
  onChange: (id: T) => void
  ariaLabel: string
}
```

Behavior requirements:
- Renders a pill container (`border-radius: var(--radius-pill)`, `background: var(--color-paper)`, `border: 1px solid var(--color-rule)`, `padding: 4px`, `display: inline-flex`).
- Each tab is a `<button role="tab" aria-selected={...}>` with inner text.
- The ACTIVE tab gets a sliding background via `motion.div` with `layoutId="segmented-indicator"` so the background animates between tabs with `springs.snappy`.
- Active tab text color: `var(--color-paper)` on `var(--color-ink)` background. Inactive: `var(--color-ink-2)` / transparent.
- Keyboard: Arrow Left / Arrow Right cycle through tabs (wrap around), Home / End jump to first/last, Enter activates, Tab moves focus out. Update `value` via `onChange` on Arrow/Home/End (Apple segmented behavior).
- ARIA: container `role="tablist" aria-label={ariaLabel}`.
- Generic `<T extends string>` so callers get typed ids (e.g., `'today' | 'week' | 'trends' | 'plan'`).

export function BentoGrid({ children, className }: { children: React.ReactNode; className?: string }): JSX.Element

Behavior requirements:

- No data fetching, no state.
