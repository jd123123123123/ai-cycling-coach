<objective>
Add per-hour intraday resolution so the user can see body battery crashes, stress spikes, and HRV overnight curves at 3-minute granularity. The five daily-level BiometricsDetail charts get a click-to-drill interaction → modal showing that day's 24h curves.

Key design decision: **lazy fetching**. We don't backfill 30 days × 3 metrics (90 Garmin API calls) at once. Only body_battery+stress+hrv ran through _safe_fetch during past syncs and we already have HRV intraday in raw_data; for body_battery & stress the intraday arrays were NOT persisted. Strategy: the endpoint serves from DB if samples exist, else fetches from Garmin on demand for that single day (3 API calls), persists, and serves.

Going forward, every new Garmin sync (hourly scheduler + initial connect + manual) will also persist the intraday arrays, so the lazy path only hits for historical gaps.

**Stretch** (only if time permits): overnight HRV ribbon chart — HRV readings every 5 min through the sleep window, with stage underlay. Backfillable from raw_data.hrvReadings we already have for all days.
</objective>

       ```python
       """Intraday sample fetcher — pulls Garmin's 3-minute body_battery, stress,
       and hrv arrays for a single day and upserts into daily_metric_samples.

       Called lazily by the GET /daily-metrics/{date}/intraday endpoint when
       samples for a day are missing, and eagerly by sync_daily_metrics after
       the daily aggregate sync.
       """
       from datetime import date as date_type, datetime, timezone
       import logging

       from sqlalchemy import select
       from sqlalchemy.dialects.postgresql import insert as pg_insert
       from sqlalchemy.ext.asyncio import AsyncSession

       from app.models.daily_metric_sample import DailyMetricSample
       from app.services.garmin_client import GarminProvider, GarminAuthError

       logger = logging.getLogger(__name__)

       METRIC_TYPES = ("body_battery", "stress", "hrv")

       def _parse_body_battery(payload: dict | list) -> list[tuple[datetime, float]]:
           """Garmin body_battery response: { 'bodyBatteryValuesArray': [[ts_ms, level, ...], ...] }
           or list wrapper around that. Extract (utc_timestamp, level) pairs.
           """
           data = payload[0] if isinstance(payload, list) and payload else payload
           if not isinstance(data, dict):
               return []
           arr = data.get("bodyBatteryValuesArray") or []
           out: list[tuple[datetime, float]] = []
           for row in arr:
               if not isinstance(row, list) or len(row) < 2:
                   continue
               ts_ms, level = row[0], row[1]
               if ts_ms is None or level is None:
                   continue
               out.append((datetime.fromtimestamp(ts_ms / 1000, tz=timezone.utc), float(level)))
           return out

       def _parse_stress(payload: dict) -> list[tuple[datetime, float]]:
           """Garmin stress response: { 'stressValuesArray': [[ts_ms, stress_level], ...] }
           stress_level of -1 or -2 means 'no reading' — skip.
           """
           if not isinstance(payload, dict):
               return []
           arr = payload.get("stressValuesArray") or []
           out: list[tuple[datetime, float]] = []
           for row in arr:
               if not isinstance(row, list) or len(row) < 2:
                   continue
               ts_ms, level = row[0], row[1]
               if ts_ms is None or level is None or level < 0:
                   continue
               out.append((datetime.fromtimestamp(ts_ms / 1000, tz=timezone.utc), float(level)))
           return out

       def _parse_hrv(payload: dict) -> list[tuple[datetime, float]]:
           """HRV response: { 'hrvReadings': [{'readingTimeGMT', 'hrvValue'}, ...] }"""
           if not isinstance(payload, dict):
               return []
           arr = payload.get("hrvReadings") or []
           out: list[tuple[datetime, float]] = []
           for r in arr:
               ts = r.get("readingTimeGMT")
               v = r.get("hrvValue")
               if ts is None or v is None:
                   continue
               try:
                   # "2026-04-19T00:03:43.0" — Garmin drops tz; treat as UTC
                   dt = datetime.fromisoformat(ts.replace("Z", "+00:00"))
                   if dt.tzinfo is None:
                       dt = dt.replace(tzinfo=timezone.utc)
                   out.append((dt, float(v)))
               except (ValueError, TypeError):
                   continue
           return out

       async def _bulk_upsert(
           db: AsyncSession,
           user_id: int,
           metric_type: str,
           points: list[tuple[datetime, float]],
       ) -> int:
           if not points:
               return 0
           rows = [
               {"user_id": user_id, "metric_type": metric_type, "recorded_at": t, "value": v}
               for t, v in points
           ]
           stmt = pg_insert(DailyMetricSample).values(rows)
           # Composite PK is (id, recorded_at); id is serial. Dedup on (user_id, metric_type, recorded_at):
           # do NOT use ON CONFLICT (needs a matching unique constraint, which hypertables can't carry
           # without partition col). Instead, delete existing same-day same-type rows, then insert.
           return len(rows)

       async def sync_intraday_samples_for_day(
           db: AsyncSession,
           client: GarminProvider,
           user_id: int,
           target_date: date_type,
       ) -> dict[str, int]:
           """Fetch body_battery + stress + hrv from Garmin for target_date,
           delete any existing samples for that (user, day, metric), bulk-insert fresh.

           Returns counts per metric. Auth errors propagate; per-metric fetch errors
           are logged and that metric is skipped.
           """
           iso = target_date.isoformat()
           result: dict[str, int] = {}

           # Delete existing same-day rows per metric (idempotent refresh)
           from datetime import time as dtime
           day_start = datetime.combine(target_date, dtime.min).replace(tzinfo=timezone.utc)
           day_end = datetime.combine(target_date, dtime.max).replace(tzinfo=timezone.utc)

           # Fetch and persist each metric
           for metric_type, fetcher in [
               ("body_battery", lambda: client.get_body_battery(iso, iso)),
               ("stress", lambda: client.get_stress_data(iso)),
               # HRV intraday uses get_hrv_data which is usually already called
               # by the daily sync — but re-fetch here so lazy loads work when
               # daily sync hasn't hit this date yet.
               ("hrv", lambda: getattr(client, "get_hrv_data", lambda _d: None)(iso)),
           ]:
               try:
                   payload = fetcher()
                   if metric_type == "body_battery":
                       points = _parse_body_battery(payload)
                   elif metric_type == "stress":
                       points = _parse_stress(payload)
                   elif metric_type == "hrv":
                       points = _parse_hrv(payload)
                   else:
                       points = []

                   # Delete & reinsert
                   from sqlalchemy import delete as sql_delete
                   await db.execute(
                       sql_delete(DailyMetricSample).where(
                           DailyMetricSample.user_id == user_id,
                           DailyMetricSample.metric_type == metric_type,
                           DailyMetricSample.recorded_at >= day_start,
                           DailyMetricSample.recorded_at <= day_end,
                       )
                   )
                   n = await _bulk_upsert(db, user_id, metric_type, points)
                   if n > 0:
                       # Insert via ORM-less bulk because the helper above just built the rows
                       rows = [
                           {"user_id": user_id, "metric_type": metric_type, "recorded_at": t, "value": v}
                           for t, v in points
                       ]
                       await db.execute(pg_insert(DailyMetricSample).values(rows))
                   result[metric_type] = n
               except GarminAuthError:
                   raise
               except Exception as exc:
                   logger.warning("Intraday %s fetch failed for user=%d date=%s: %s", metric_type, user_id, iso, exc)
                   result[metric_type] = 0

           await db.flush()
           return result
       ```

       *Design note:* I wrote `_bulk_upsert` wonky — it builds rows but the actual insert happens in the caller path. Simplify by moving the insert inline (single SQL execute via pg_insert). Consolidate into one clean function. The executor should simplify during implementation.

       ```python
       @router.get("/{iso_date}/intraday", response_model=IntradayDayResponse)
       async def get_intraday_for_day(
           iso_date: date,
           current_user: User = Depends(get_current_active_user),
           db: AsyncSession = Depends(get_db_for_user),
       ) -> IntradayDayResponse:
           from datetime import datetime, time as dtime, timezone
           from sqlalchemy import select as sql_select
           from app.models.daily_metric_sample import DailyMetricSample
           from app.models.garmin_credential import GarminCredential
           from app.services.garmin_sync_service import create_garmin_client_for_user
           from app.services.intraday_sync_service import sync_intraday_samples_for_day

           day_start = datetime.combine(iso_date, dtime.min).replace(tzinfo=timezone.utc)
           day_end = datetime.combine(iso_date, dtime.max).replace(tzinfo=timezone.utc)

           async def load() -> dict[str, list[DailyMetricSample]]:
               result = await db.execute(
                   sql_select(DailyMetricSample)
                   .where(
                       DailyMetricSample.user_id == current_user.id,
                       DailyMetricSample.recorded_at >= day_start,
                       DailyMetricSample.recorded_at <= day_end,
                   )
                   .order_by(DailyMetricSample.recorded_at.asc())
               )
               rows = result.scalars().all()
               grouped: dict[str, list[DailyMetricSample]] = {"body_battery": [], "stress": [], "hrv": [], "hr": []}
               for r in rows:
                   grouped.setdefault(r.metric_type, []).append(r)
               return grouped

           cached = await load()
           has_anything = any(len(samples) > 0 for samples in cached.values())
           fetched_live = False

           if not has_anything:
               # Lazy fetch from Garmin for this specific day
               cred_result = await db.execute(
                   sql_select(GarminCredential).where(GarminCredential.user_id == current_user.id)
               )
               credential = cred_result.scalar_one_or_none()
               if credential is not None and credential.sync_enabled:
                   try:
                       client = create_garmin_client_for_user(credential)
                       await sync_intraday_samples_for_day(db, client, current_user.id, iso_date)
                       await db.commit()
                       cached = await load()
                       fetched_live = True
                   except Exception as exc:
                       # Don't 500; return empty payload and let the UI handle it
                       import logging
                       logging.getLogger(__name__).warning(
                           "Lazy intraday fetch failed: %s", exc
                       )

           def to_points(samples: list[DailyMetricSample]) -> list[dict]:
               return [{"t": s.recorded_at, "v": s.value} for s in samples]

           return IntradayDayResponse(
               date=iso_date,
               body_battery=to_points(cached.get("body_battery", [])),
               stress=to_points(cached.get("stress", [])),
               hrv=to_points(cached.get("hrv", [])),
               hr=to_points(cached.get("hr", [])),
               fetched_live=fetched_live,
           )
       ```
