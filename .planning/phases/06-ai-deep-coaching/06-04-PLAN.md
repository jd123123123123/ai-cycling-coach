```typescript
const MAX_CONTEXT_TURNS = 10;

function toBackendHistory(messages: Message[], newUserMessage: string): BackendMessage[] {
  // Keep only user+assistant messages (drop tool cards — they're UI-only)
  const conversational = messages.filter(m => m.role === 'user' || m.role === 'assistant');
  const recent = conversational.slice(-MAX_CONTEXT_TURNS);
  return [
    ...recent.map(m => ({ role: m.role, content: m.content })),
    { role: 'user', content: newUserMessage },
  ];
}
```

1. `backend/app/services/workout_modifier.py` — add `revert_from_audit`:

```python
async def revert_from_audit(audit_row: CoachBrief, db: AsyncSession) -> ModificationResult:
    """
    Restore a PlannedWorkout to its pre-modification state using the 'before'
    snapshot stored in audit_row.evidence.modification.before.
    Creates a new audit row with verdict='modification_undo'.
    """
    mod = audit_row.evidence.get("modification")
    if not mod:
        raise ValueError("Audit row lacks evidence.modification; cannot undo")

    before = mod["before"]
    date = mod["date"]

    pw = await _get_planned_workout(audit_row.user_id, date, db)
    if pw is None:
        raise ValueError(f"PlannedWorkout for {date} no longer exists; cannot undo")

    # Capture CURRENT state for the undo audit's 'before' (post-modification state)
    current = {
        "template_id": pw.template_id,
        "duration_min": pw.duration_min,
    }

    # Restore PW to pre-modification state
    pw.template_id = before["template_id"]
    pw.duration_min = before["duration_min"]
    pw.blocks = parametrize_template(before["template_id"], _ftp_for(audit_row.user_id, db),
                                     duration_override=before["duration_min"])["blocks"]
    pw.blocks_source = "user_undo"
    await db.flush()

    # Create undo audit row (append-only — never mutate the original audit)
    undo_audit = CoachBrief(
        user_id=audit_row.user_id,
        created_at=datetime.utcnow(),
        verdict="modification_undo",
        rationale=f"Undid modification on {date}",
        evidence={
            "modification_undo": {
                "original_audit_id": str(audit_row.id),
                "reverted_from": current,
                "restored_to": before,
                "date": date,
            },
        },
    )
    db.add(undo_audit)
    await db.commit()

    return ModificationResult(
        template_id=before["template_id"],
        duration_min=before["duration_min"],
        blocks=pw.blocks,
        indoor=pw.indoor,
        rails_downgraded=False,
        reason=f"Undid modification on {date}",
        before=current,
    )
```

Handler:

```python
async def _handle_undo_last_modification(tool_input, user_id, db):
    # Find most recent CoachBrief with verdict='modification' in last 60 min
    # that has not already been undone (no subsequent modification_undo referencing it)
    cutoff = datetime.utcnow() - timedelta(hours=1)
    stmt = (select(CoachBrief)
            .where(CoachBrief.user_id == user_id,
                   CoachBrief.verdict == "modification",
                   CoachBrief.created_at >= cutoff)
            .order_by(CoachBrief.created_at.desc()))
    row = (await db.execute(stmt)).scalars().first()
    if not row:
        return {"found": False, "reason": "No modifications in the last hour to undo"}

    # Check if already undone
    already_stmt = (select(CoachBrief)
                    .where(CoachBrief.user_id == user_id,
                           CoachBrief.verdict == "modification_undo",
                           CoachBrief.evidence["modification_undo"]["original_audit_id"].astext == str(row.id)))
    if (await db.execute(already_stmt)).scalars().first():
        return {"found": False, "reason": "That modification has already been undone"}

    result = await revert_from_audit(row, db)
    return {
        "success": True,
        "restored_template_id": result.template_id,
        "restored_duration_min": result.duration_min,
        "reverted_from": result.before,
        "date": row.evidence["modification"]["date"],
    }

TOOL_REGISTRY["undo_last_modification"] = _handle_undo_last_modification
```
