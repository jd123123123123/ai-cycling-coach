       ```ts
       export interface PlannedWorkout {
         id: number
         user_id: number
         date: string                 // ISO yyyy-mm-dd
         sport: 'cycling' | 'running' | 'gym'
         title: string
         duration_seconds_target: number | null
         description: string | null
         interval_structure: { intervals: Array<{ duration_seconds: number; power_target_pct_ftp: number | null; zone: number }> } | null
         created_at: string
       }

       export interface PlannedWorkoutListResponse {
         planned_workouts: PlannedWorkout[]
         count: number
       }

       export interface PlannedWorkoutCreateBody {
         date: string                 // yyyy-mm-dd
         sport: 'cycling' | 'running' | 'gym'
         title: string
         duration_seconds_target?: number | null
         description?: string | null
         interval_structure?: PlannedWorkout['interval_structure']
       }

       export interface PlannedWorkoutUpdateBody {
         date?: string
         sport?: 'cycling' | 'running' | 'gym'
         title?: string
         duration_seconds_target?: number | null
         description?: string | null
         interval_structure?: PlannedWorkout['interval_structure']
       }

       export async function fetchPlannedWorkouts(start: Date, end: Date): Promise<PlannedWorkoutListResponse> {
         const toIsoDate = (d: Date) => `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`
         const res = await apiClient.get<PlannedWorkoutListResponse>('/planned-workouts', {
           params: { start_date: toIsoDate(start), end_date: toIsoDate(end) },
         })
         return res.data
       }

       export async function createPlannedWorkout(body: PlannedWorkoutCreateBody): Promise<PlannedWorkout> {
         const res = await apiClient.post<PlannedWorkout>('/planned-workouts', body)
         return res.data
       }

       export async function updatePlannedWorkout(id: number, body: PlannedWorkoutUpdateBody): Promise<PlannedWorkout> {
         const res = await apiClient.patch<PlannedWorkout>(`/planned-workouts/${id}`, body)
         return res.data
       }

       export async function deletePlannedWorkout(id: number): Promise<void> {
         await apiClient.delete(`/planned-workouts/${id}`)
       }
       ```

    3. Create `frontend/src/hooks/usePlannedWorkouts.ts`:

       ```ts
       import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
       import {
         fetchPlannedWorkouts, createPlannedWorkout, updatePlannedWorkout, deletePlannedWorkout,
         type PlannedWorkout, type PlannedWorkoutListResponse,
         type PlannedWorkoutCreateBody, type PlannedWorkoutUpdateBody,
       } from '../api/dashboard'

       export function usePlannedWorkouts(start: Date, end: Date) {
         return useQuery<PlannedWorkoutListResponse>({
           queryKey: ['planned-workouts', start.toDateString(), end.toDateString()],
           queryFn: () => fetchPlannedWorkouts(start, end),
         })
       }

       export function useCreatePlannedWorkout() {
         const qc = useQueryClient()
         return useMutation({
           mutationFn: (body: PlannedWorkoutCreateBody) => createPlannedWorkout(body),
           onSuccess: () => { qc.invalidateQueries({ queryKey: ['planned-workouts'] }) },
         })
       }

       // Optimistic update — critical for drag-drop UX. Rolls back on error.
       export function useUpdatePlannedWorkout() {
         const qc = useQueryClient()
         return useMutation({
           mutationFn: ({ id, body }: { id: number; body: PlannedWorkoutUpdateBody }) => updatePlannedWorkout(id, body),
           onMutate: async ({ id, body }) => {
             await qc.cancelQueries({ queryKey: ['planned-workouts'] })
             const prev = qc.getQueriesData<PlannedWorkoutListResponse>({ queryKey: ['planned-workouts'] })
             for (const [key, data] of prev) {
               if (!data) continue
               qc.setQueryData<PlannedWorkoutListResponse>(key, {
                 ...data,
                 planned_workouts: data.planned_workouts.map((pw: PlannedWorkout) =>
                   pw.id === id ? { ...pw, ...body } : pw,
                 ),
               })
             }
             return { prev }
           },
           onError: (_err, _vars, ctx) => {
             if (ctx?.prev) {
               for (const [key, data] of ctx.prev) qc.setQueryData(key, data)
             }
           },
           onSettled: () => { qc.invalidateQueries({ queryKey: ['planned-workouts'] }) },
         })
       }

       export function useDeletePlannedWorkout() {
         const qc = useQueryClient()
         return useMutation({
           mutationFn: (id: number) => deletePlannedWorkout(id),
           onSuccess: () => { qc.invalidateQueries({ queryKey: ['planned-workouts'] }) },
         })
       }
       ```
