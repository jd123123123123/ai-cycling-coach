**Create `frontend/src/lib/coach-chat-stream.ts`:**

```typescript
export type ChatStreamEvent =
  | { type: 'content', delta: string }
  | { type: 'replace', text: string }
  | { type: 'tool_use_start', name: string, input: Record<string, unknown> }
  | { type: 'tool_use_result', name: string, result: Record<string, unknown> }
  | { type: 'done' }
  | { type: 'error', message: string };

export async function* parseCoachStream(
  response: Response
): AsyncGenerator<ChatStreamEvent> {
  if (!response.body) { yield { type: 'error', message: 'No response body' }; return; }
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  while (true) {
    const { value, done } = await reader.read();
    if (done) { yield { type: 'done' }; return; }
    buffer += decoder.decode(value, { stream: true });

    // SSE frames are separated by '\n\n'; each frame has 'event: X\ndata: Y'
    let idx;
    while ((idx = buffer.indexOf('\n\n')) !== -1) {
      const frame = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 2);
      const event = parseFrame(frame);
      if (event) yield event;
    }
  }
}

function parseFrame(frame: string): ChatStreamEvent | null {
  const lines = frame.split('\n');
  let event = 'content';
  let dataStr = '';
  for (const line of lines) {
    if (line.startsWith('event:')) event = line.slice(6).trim();
    else if (line.startsWith('data:')) dataStr += line.slice(5).trim();
  }
  try {
    const data = JSON.parse(dataStr);
    switch (event) {
      case 'content': return { type: 'content', delta: data.delta };
      case 'replace': return { type: 'replace', text: data.text };
      case 'tool_use_start': return { type: 'tool_use_start', name: data.name, input: data.input };
      case 'tool_use_result': return { type: 'tool_use_result', name: data.name, result: data.result };
      default: return null;
    }
  } catch {
    return { type: 'error', message: `Malformed frame: ${frame.slice(0, 80)}` };
  }
}
```

**Create `frontend/src/hooks/useChatStream.ts`:**

```typescript
export type ChatState = {
  messages: Array<Message>;
  streaming: boolean;
  currentToolUse: { name: string; input: Record<string, unknown> } | null;
  toolUseLog: Array<{ name: string; input: Record<string, unknown>; result?: Record<string, unknown> }>;
  error: string | null;
};

export type ChatStreamOptions = {
  language: 'simple' | 'technical';
  onModification?: (result: WorkoutModificationResult) => void;
};

export function useChatStream(opts: ChatStreamOptions) {
  const [state, setState] = useState<ChatState>(initialState);
  const queryClient = useQueryClient();

  const send = async (userMessage: string) => {
    // append user message; set streaming true
    setState(s => ({ ...s, messages: [...s.messages, { role: 'user', content: userMessage }], streaming: true, error: null }));

    const response = await fetch('/coach/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
      body: JSON.stringify({ messages: toBackendHistory(state.messages, userMessage), language: opts.language }),
    });
    if (response.status === 503) { setState(s => ({ ...s, streaming: false, error: 'Coach upgrade required (ANTHROPIC_API_KEY)' })); return; }
    if (!response.ok) { setState(s => ({ ...s, streaming: false, error: `HTTP ${response.status}` })); return; }

    let assistantText = '';
    // Append an empty assistant message to append into
    setState(s => ({ ...s, messages: [...s.messages, { role: 'assistant', content: '' }] }));

    for await (const evt of parseCoachStream(response)) {
      if (evt.type === 'content') {
        assistantText += evt.delta;
        setState(s => replaceLastAssistant(s, assistantText));
      } else if (evt.type === 'replace') {
        assistantText = evt.text;
        setState(s => replaceLastAssistant(s, assistantText));
      } else if (evt.type === 'tool_use_start') {
        setState(s => ({ ...s, currentToolUse: { name: evt.name, input: evt.input },
                        toolUseLog: [...s.toolUseLog, { name: evt.name, input: evt.input }] }));
      } else if (evt.type === 'tool_use_result') {
        setState(s => ({ ...s, currentToolUse: null,
                        toolUseLog: s.toolUseLog.map((t, i) => i === s.toolUseLog.length - 1 ? { ...t, result: evt.result } : t),
                        messages: appendToolResultCard(s.messages, evt.name, evt.result) }));
        // Invalidate caches for write-tool results
        if (evt.name === 'modify_planned_workout' && evt.result.success) {
          queryClient.invalidateQueries({ queryKey: ['planned-workouts'] });
          queryClient.invalidateQueries({ queryKey: ['coach', 'brief'] });
          opts.onModification?.(evt.result as WorkoutModificationResult);
        }
      } else if (evt.type === 'error') {
        setState(s => ({ ...s, error: evt.message }));
      } else if (evt.type === 'done') {
        setState(s => ({ ...s, streaming: false, currentToolUse: null }));
      }
    }
  };

  return { ...state, send };
}
```

**Design notes:**
- Tool_use cards are inserted into the `messages` array as a special message type (`{ role: 'tool', toolName, result }`) so they render inline in chat history naturally
- React Query invalidation happens on tool_use_result for write tools (`modify_planned_workout`); read tools don't invalidate anything
- `currentToolUse` drives the "Analyzing your training data..." indicator
- `toolUseLog` is the data source for TransparencyDrawer's "Tools Used" section
- No tests for this file (it's a thin wrapper around fetch + parser — UI tests in task 3 cover it)
